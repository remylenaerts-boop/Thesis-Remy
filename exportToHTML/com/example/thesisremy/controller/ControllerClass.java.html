package com.example.thesisremy.controller;



import com.example.thesisremy.ServicesAndComponents.ValueBroadcastService;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.servlet.mvc.method.annotation.SseEmitter;


/**
 *
 * <p> This classThis classThis classThis classThis class</p>
 */

@RestController //marks this class as a HTTP request handler
public class ControllerClass {
    private final ValueBroadcastService valueBroadcastService;

    public ControllerClass(ValueBroadcastService valueBroadcastService) {
        this.valueBroadcastService = valueBroadcastService;
    }

    @GetMapping("/stream")
    public SseEmitter stream() {
        return valueBroadcastService.addEmitter();
    }
}
